# The cast rig, release 1 (0.11.0): the engine with the mangaka and the assistants

> **For agentic workers:** REQUIRED SUB-SKILL: Use the subagent-driven-development skill (recommended) or the executing-plans skill to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Draw the mangaka and the three assistants as 32x48 chibi characters composed from a rig of hand-drawn parts and
key frames, painted once per palette into a per-session sprite cache and animated by a pure player (base loops,
transitions, seeded idle moments, reactions to game events), with the spots re-placed for the bigger figures; the
editor and the cat keep their 0.10 frames, drawn through the same cache at their old sizes.

**Architecture:** Pure modules carry the engine: `rig/` (types, the assembler, the head, the parts, the key frames, the
pose and animation tables, `rigFrame`), `spriteCache.ts`, `animation.ts`, `idleSchedule.ts`, `reactions.ts` and the
driver `castPlay.ts`, which turns each actor's state into a pose on the actor (`Actor.pose`) every loop step.
`render.ts` draws every character with one `drawImage` from the cache and keeps its code-drawn extras (shadows, the
blanket, glyphs); `StudioScene.tsx` owns the scene clock, feeds the driver and turns store transitions into the
snapshot's one new field, `reaction`. A switch (`RIG_LIVE`) keeps the 0.10 frames drawing until each character's
set is complete; a dev build's `?rig` URL turns it on early for the art tasks' in-scene check.

**Tech Stack:** React 19, TypeScript (`erasableSyntaxOnly`), Vite, Vitest (node and happy-dom), Zustand, oxlint;
`scripts/spriteSheet.ts` (vite-node + sharp) for the contact sheet.

**Spec:** `docs/plans/2026-10-09-cast-rig-design.md` (release 1 only; 0.12.0 the editor and 0.13.0 the cat are out of
scope here).

   the character cannot play in its state is dropped; one it can play cuts an idle moment short and waits behind a

   opens on the panic (640 ms in every 3640 ms, 0.10's 40 frames in 240), so the panic recurs as today and reduced
   motion still shows it.
8. **Reduced motion:** every character holds its base loop's first step (the driver, and render's fallback for an
   actor with no pose); no transition, idle moment or reaction plays.
9. **The editor and the cat** keep their 0.10 maps and frame-counter timing, drawn through the cache at 24x32 and

```ts
// The cast rig's types (0.11.0): parts with named anchors, poses that join them, the head's face slots, and the frame
// the sprite cache draws. Types and two small tables; no DOM and no sim, so scripts/ can load the rig.
import type { SpriteMap } from '../spriteCore'

export type { SpriteMap }

/** A mood the head shows (mood.ts `moodOf` picks it and re-exports this type). */
export type Mood = 'neutral' | 'happy' | 'tired' | 'surprised' | 'sleepy'

/** One frame the cast draws: its id (the sprite cache names its surfaces by it) and its map. */
export interface CastFrame { readonly id: string; readonly map: SpriteMap }

/** Where two parts join: the torso has all five, a head its neck, an arm its shoulder and hand, a leg its hip, a prop its hand. */
export type AnchorName = 'neck' | 'shoulderL' | 'shoulderR' | 'hipL' | 'hipR' | 'hand'

/** A cell inside a part's map: column x, row y. */
export interface Point { readonly x: number; readonly y: number }

/** One hand-drawn piece of a character: a string map in PAL keys and its anchors (each inside the map, on a painted cell). */
export interface Part {
  readonly map: SpriteMap
  readonly anchors: Readonly<Partial<Record<AnchorName, Point>>>
}

/** A rectangle inside a part: its top-left cell, width and height. */
export interface Slot { readonly x: number; readonly y: number; readonly w: number; readonly h: number }

/** The head: a part with three face slots (the eyes and the mouth in the face, the mark beside the hair's top-right). */
export interface HeadPart extends Part {
  readonly slots: { readonly eyes: Slot; readonly mouth: Slot; readonly mark: Slot }
}

export type EyesId = 'open' | 'blink' | 'closed' | 'ring' | 'squint' | 'happy'
export type MouthId = 'closed' | 'open' | 'smile' | 'frown' | 'o'
export type MarkId = 'none' | 'sweat' | 'sparkle' | 'bang' | 'zee'

/** A face: one drawing per slot. */
export interface Face { readonly eyes: EyesId; readonly mouth: MouthId; readonly mark: MarkId }

/** Every face drawing: the eyes and the mouths fill their slot whole (no '.'); a mark leaves its '.' cells clear. */
export interface FaceArt {
  readonly eyes: Readonly<Record<EyesId, SpriteMap>>
  readonly mouths: Readonly<Record<MouthId, SpriteMap>>
  readonly marks: Readonly<Record<Exclude<MarkId, 'none'>, SpriteMap>>
}

/** The layers, back to front. */
export type Layer = 'armBack' | 'legBack' | 'legFront' | 'torso' | 'head' | 'armFront' | 'prop'
export const LAYERS: readonly Layer[] = ['armBack', 'legBack', 'legFront', 'torso', 'head', 'armFront', 'prop']

/** The anchor each layer joins on: its parent's anchor of the same name (the torso's; a prop's parent is the front arm). */
export const JOIN: Readonly<Record<Exclude<Layer, 'torso'>, AnchorName>> = {
  armBack: 'shoulderR', legBack: 'hipR', legFront: 'hipL', head: 'neck', armFront: 'shoulderL', prop: 'hand',
}

/** A part placed by its join anchor, nudged by (dx, dy) when a pose needs it (a breath, a glance). */
export interface Place { readonly part: string; readonly dx?: number; readonly dy?: number }

/** A pose built from parts: the frame's size, the torso's top-left in it, every other layer by its anchor. */
export interface PartPose {
  readonly kind: 'parts'
  readonly w: number
  readonly h: number
  readonly torso: { readonly part: string; readonly x: number; readonly y: number }
  readonly head: Place
  readonly armBack?: Place
  readonly legBack?: Place
  readonly legFront?: Place
  readonly armFront?: Place
  readonly prop?: Place
  /** Slots the pose fixes over the mood's face (a blink's eyes, a sip's closed eyes). */
  readonly face?: Partial<Face>
}

/** A pose the parts cannot make, hand-drawn whole, its face baked in. */
export interface KeyPose { readonly kind: 'key'; readonly map: SpriteMap }

export type Pose = PartPose | KeyPose

/** The parts a pose may name, the head among them. */
export type PartLib = Readonly<Record<string, Part | HeadPart>>
```

```ts
// The assembler: a pose's parts composed into one frame, back to front, each placed so its anchor lands on its
// parent's; the outline closed where parts meet; the face stamped into the head's slots. Pure: no DOM, no clock.
import { PAL, frameRules } from '../spriteCore'
import { JOIN, LAYERS } from './types'
import type { AnchorName, Face, FaceArt, HeadPart, Layer, Part, PartLib, PartPose, Point, SpriteMap } from './types'

/** Whether a part is the head (it has face slots). */
export function isHead(p: Part | HeadPart): p is HeadPart {
  return 'slots' in p
}

/** A drawing's painted cells over a grid, its top-left at (x, y); '.' leaves the grid; a cell outside it throws. */
function paint(grid: string[][], map: SpriteMap, x: number, y: number, what: string): void {
  map.forEach((row, r) => {
    for (let c = 0; c < row.length; c++) {
      if (row[c] === '.') continue
      const gx = x + c
      const gy = y + r
      if (gy < 0 || gy >= grid.length || gx < 0 || gx >= grid[gy].length) {
        throw new Error(`${what}: cell ${c},${r} lands outside the frame at ${gx},${gy}`)
      }
      grid[gy][gx] = row[c]
    }
  })
}

/**
 * The outline closed: every painted cell but L with a clear cell beside it (left, right, above or below; the frame's
 * edge counts as clear) becomes L, judged against the map as it was, so a new L never spreads.
 */
export function closeSilhouette(rows: SpriteMap): SpriteMap {
  const clear = (x: number, y: number) => y < 0 || y >= rows.length || x < 0 || x >= rows[y].length || rows[y][x] === '.'
  return rows.map((row, y) => row.replace(/./g, (ch, x: number) =>
    ch === '.' || ch === 'L' ? ch : clear(x - 1, y) || clear(x + 1, y) || clear(x, y - 1) || clear(x, y + 1) ? 'L' : ch))
}

/** The head's map with the face's eyes and mouth in their slots (each drawing fills its slot whole). */
export function stampFace(head: HeadPart, art: FaceArt, face: Face): SpriteMap {
  const grid = head.map.map((row) => row.split(''))
  paint(grid, art.eyes[face.eyes], head.slots.eyes.x, head.slots.eyes.y, `eyes ${face.eyes}`)
  paint(grid, art.mouths[face.mouth], head.slots.mouth.x, head.slots.mouth.y, `mouth ${face.mouth}`)
  return grid.map((row) => row.join(''))
}

function partOf(lib: PartLib, name: string, layer: Layer): Part | HeadPart {
  const p = lib[name]
  if (!p) throw new Error(`${layer}: no part ${name}`)
  return p
}

function anchorOf(part: Part, name: AnchorName, what: string): Point {
  const a = part.anchors[name]
  if (!a) throw new Error(`${what} has no ${name} anchor`)
  return a
}

/** One placed layer: its part and the frame cell its top-left lands on. */
export interface Placed { readonly part: Part | HeadPart; readonly at: Point }

/** Where every layer the pose names lands: the torso at its own cell, each other part by its join anchor. */
export function placeLayers(pose: PartPose, lib: PartLib): Map<Layer, Placed> {
  const out = new Map<Layer, Placed>()
  out.set('torso', { part: partOf(lib, pose.torso.part, 'torso'), at: { x: pose.torso.x, y: pose.torso.y } })
  for (const layer of LAYERS) {
    if (layer === 'torso') continue
    const place = pose[layer]
    if (!place) continue
    const part = partOf(lib, place.part, layer)
    const parentLayer: Layer = layer === 'prop' ? 'armFront' : 'torso'
    const parent = out.get(parentLayer)
    if (!parent) throw new Error(`${layer} ${place.part} needs the ${parentLayer} layer`)
    const name = JOIN[layer]
    const pa = anchorOf(parent.part, name, `${parentLayer} ${pose[parentLayer]?.part}`)
    const ca = anchorOf(part, name, `${layer} ${place.part}`)
    out.set(layer, { part, at: { x: parent.at.x + pa.x - ca.x + (place.dx ?? 0), y: parent.at.y + pa.y - ca.y + (place.dy ?? 0) } })
  }
  return out
}

/**
 * A pose composed into one w x h frame: the layers back to front (a later layer paints over an earlier one, '.' is
 * see-through), the head with the face's eyes and mouth in its slots, the outline closed where parts meet
 * (`closeSilhouette`), then the face's mark stamped into the head's mark slot, whose cells must be clear.
 */
export function assemblePose(pose: PartPose, lib: PartLib, art: FaceArt, face: Face): SpriteMap {
  const placed = placeLayers(pose, lib)
  const grid = Array.from({ length: pose.h }, () => Array.from({ length: pose.w }, () => '.'))
  for (const layer of LAYERS) {
    const p = placed.get(layer)
    if (!p) continue
    let map = p.part.map
    if (layer === 'head') {
      if (!isHead(p.part)) throw new Error(`head ${pose.head.part} has no face slots`)
      map = stampFace(p.part, art, face)
    }
    paint(grid, map, p.at.x, p.at.y, `${layer} ${pose[layer]?.part}`)
  }
  const closed = closeSilhouette(grid.map((row) => row.join('')))
  const head = placed.get('head')
  if (face.mark === 'none' || !head || !isHead(head.part)) return closed
  const out = closed.map((row) => row.split(''))
  const slot = head.part.slots.mark
  art.marks[face.mark].forEach((row, r) => {
    for (let c = 0; c < row.length; c++) {
      if (row[c] === '.') continue
      const x = head.at.x + slot.x + c
      const y = head.at.y + slot.y + r
      if (y < 0 || y >= out.length || x < 0 || x >= out[y].length) throw new Error(`mark ${face.mark} lands outside the frame at ${x},${y}`)
      if (out[y][x] !== '.') throw new Error(`mark ${face.mark} lands on a painted cell at ${x},${y}`)
      out[y][x] = row[c]
    }
  })
  return out.map((row) => row.join(''))
}

/**
 * What is wrong with a part, [] when nothing: every row one width, every key in `pal`, and each anchor in `need`
 * present, inside the map and on a painted cell (so two parts always meet on pixels).
 */
export function partRules(part: Part, need: readonly AnchorName[], pal: Record<string, string> = PAL): string[] {
  const out: string[] = []
  const w = part.map[0]?.length ?? 0
  part.map.forEach((row, r) => {
    if (row.length !== w) out.push(`row ${r} is ${row.length} wide, not ${w}`)
    for (const ch of row) if (ch !== '.' && !(ch in pal)) out.push(`row ${r} has key ${ch}`)
  })
  for (const name of need) {
    const a = part.anchors[name]
    if (!a) out.push(`no ${name} anchor`)
    else if (a.y < 0 || a.y >= part.map.length || a.x < 0 || a.x >= w) out.push(`${name} at ${a.x},${a.y} is outside the part`)
    else if ((part.map[a.y][a.x] ?? '.') === '.') out.push(`${name} at ${a.x},${a.y} is on a clear cell`)
  }
  return out
}

/**
 * What stops a part pose, [] when nothing: each face tried, every refusal of the assembler said once, and the frame
 * rules at the pose's size for each face with no mark (a mark is a sticker beside the hair, outside the outline rule,
 * as the 0.10 MARKS were).
 */
export function poseRules(
  pose: PartPose, lib: PartLib, art: FaceArt, faces: readonly Face[], pal: Record<string, string> = PAL,
): string[] {
  const out = new Set<string>()
  for (const face of faces) {
    try {
      const map = assemblePose(pose, lib, art, face)
      if (face.mark === 'none') for (const r of frameRules(map, pose.w, pose.h, pal)) out.add(r)
    } catch (e) {
      out.add((e as Error).message)
    }
  }
  return [...out]
}
```

```ts
// The sprite cache (0.11.0): each cast frame painted once per palette (and once more mirrored) into its own offscreen
// surface, built on first use and kept for the session, so render draws a character with one drawImage. A surface is
// an OffscreenCanvas, else a <canvas>, else (no canvas at all: node tests) a stand-in carrying its id and size, which
// the recording test context names in `images`. A hit allocates nothing: two Map lookups by the frame's stored id and
// a constant palette id.
import { ASSISTANT_PALS, OUTFIT_PALS, PAL, drawSprite, outfitOf } from './sprites'
import type { CastFrame } from './rig/types'

export type { CastFrame }

export type PaletteId = 'pal' | 'cardigan' | 'blazer' | 'look0' | 'look1' | 'look2' | 'look3' | 'look4' | 'look5'

/** Every palette a cast frame is drawn in: PAL (the hoodie, the editor, the cat), the other two outfits, the six looks. */
export const CAST_PALETTES: Readonly<Record<PaletteId, Record<string, string>>> = {
  pal: PAL, cardigan: OUTFIT_PALS.cardigan, blazer: OUTFIT_PALS.blazer,
  look0: ASSISTANT_PALS[0], look1: ASSISTANT_PALS[1], look2: ASSISTANT_PALS[2],
  look3: ASSISTANT_PALS[3], look4: ASSISTANT_PALS[4], look5: ASSISTANT_PALS[5],
}

const LOOK_IDS: readonly PaletteId[] = ['look0', 'look1', 'look2', 'look3', 'look4', 'look5']

/** The mangaka's palette for a level (`outfitOf`; the hoodie is PAL). */
export function outfitPalette(level: number): PaletteId {
  const o = outfitOf(level)
  return o === 'hoodie' ? 'pal' : o
}

/** An assistant's palette for its look, wrapping as ASSISTANT_PALS does. */
export function lookPalette(look: number): PaletteId {
  return LOOK_IDS[look % LOOK_IDS.length]
}

/** Where the cache paints one frame: the image render draws, and the context that paints it (null: nothing to paint into). */
export interface Surface { readonly image: CanvasImageSource; readonly ctx: CanvasRenderingContext2D | null }
export type SurfaceFactory = (w: number, h: number, id: string) => Surface

/** A surface's id: `cast/<frame id>/<palette>`, `/r` added when mirrored. */
export function castSurfaceId(frameId: string, pal: PaletteId, flip: boolean): string {
  return `cast/${frameId}/${pal}${flip ? '/r' : ''}`
}

/**
 * An OffscreenCanvas, else a <canvas> (also when an OffscreenCanvas gives no 2D context, as Safari 16.0-16.3's do),
 * else a stand-in image (its id and size, nothing painted).
 */
export function defaultSurface(w: number, h: number, id: string): Surface {
  if (typeof OffscreenCanvas !== 'undefined') {
    const c = new OffscreenCanvas(w, h)
    const ctx = c.getContext('2d') as unknown as CanvasRenderingContext2D | null
    if (ctx) return { image: c, ctx }
  }
  if (typeof document !== 'undefined') {
    const c = document.createElement('canvas')
    c.width = w
    c.height = h
    return { image: c, ctx: c.getContext('2d') }
  }
  return { image: { id, width: w, height: h } as unknown as CanvasImageSource, ctx: null }
}

export interface SpriteCache {
  /** The frame in the palette, mirrored when `flip`: painted on first use, the same image every time after. */
  image(frame: CastFrame, pal: PaletteId, flip: boolean): CanvasImageSource
  /** Surfaces built so far. */
  readonly size: number
}

export function createSpriteCache(make: SurfaceFactory = defaultSurface): SpriteCache {
  const byFrame = new Map<string, Map<PaletteId, [CanvasImageSource | null, CanvasImageSource | null]>>()
  let size = 0
  return {
    image(frame, pal, flip) {
      let byPal = byFrame.get(frame.id)
      if (!byPal) {
        byPal = new Map()
        byFrame.set(frame.id, byPal)
      }
      let pair = byPal.get(pal)
      if (!pair) {
        pair = [null, null]
        byPal.set(pal, pair)
      }
      const k = flip ? 1 : 0
      const have = pair[k]
      if (have) return have
```

```ts
// The animation player (0.11.0), pure: an animation is a list of steps (a pose and how long it shows), looping or played
// once; a player holds a character's base loop, the one-shot playing over it (a transition, an idle moment or a
// reaction) and the one-shots waiting. Time is the scene's own clock in ms (StudioScene); nothing here reads a clock,
// and a step where nothing ends returns the same state object.

/** How a character is posed: a base change between two postures drops the moments drawn for the old one. */
export type Posture = 'stand' | 'walk' | 'seat' | 'bed'

export interface Step<P extends string = string> { readonly pose: P; readonly ms: number }

export interface Anim<P extends string = string> {
  readonly id: string
  readonly posture: Posture
  readonly loop: boolean
  readonly steps: readonly Step<P>[]
}

/** An animation's length in ms: the sum of its steps. */
export function animLength(a: Anim): number {
  let n = 0
  for (const s of a.steps) n += s.ms
  return n
}

/** The step showing `t` ms in (before 0, the first): a loop wraps; a one-shot past its end holds its last step. */
export function stepAt<P extends string>(a: Anim<P>, t: number): Step<P> {
  const len = animLength(a)
  let at = t <= 0 || len <= 0 ? 0 : a.loop ? t % len : Math.min(t, len - 1)
  for (const s of a.steps) {
    if (at < s.ms) return s
    at -= s.ms
  }
  return a.steps[a.steps.length - 1]
}

/** A one-shot playing over the base: the animation and when it started. */
export interface Overlay<P extends string = string> { readonly anim: Anim<P>; readonly at: number }

export interface PlayerState<P extends string = string> {
  /** The base loop (the actor state's) and when it started, so a loop keeps its phase under a moment. */
  readonly base: Anim<P>
  readonly baseAt: number
  /** The one-shot playing now, or null. */
  readonly over: Overlay<P> | null
  /** One-shots waiting, each starting when the one before it ends. */
  readonly queue: readonly Anim<P>[]
}

/** The most one-shots that may wait; one more is dropped. */
export const QUEUE_MAX = 2

export function startPlayer<P extends string>(base: Anim<P>, now: number): PlayerState<P> {
  return { base, baseAt: now, over: null, queue: [] }
}

/**
 * A new base loop (the actor's state changed). With a transition, it plays first and the base starts as it ends.
 * Within one posture (drawing to idle at the desk) the moment playing and the queue are kept; across postures they are
 * dropped, drawn as they were for the old one. The same base and no transition: nothing changes.
 */
export function switchBase<P extends string>(
  p: PlayerState<P>, base: Anim<P>, transition: Anim<P> | null, now: number,
): PlayerState<P> {
  if (base === p.base && !transition) return p
  if (transition) return { base, baseAt: now + animLength(transition), over: { anim: transition, at: now }, queue: [] }
  if (base.posture === p.base.posture) return { ...p, base, baseAt: now }
  return { base, baseAt: now, over: null, queue: [] }
}

/** A one-shot over the base: at once when nothing plays, else after what waits; a full queue drops it. */
export function enqueue<P extends string>(p: PlayerState<P>, anim: Anim<P>, now: number): PlayerState<P> {
  if (!p.over) return { ...p, over: { anim, at: now } }
  if (p.queue.length >= QUEUE_MAX) return p
  return { ...p, queue: [...p.queue, anim] }
}

/** A one-shot that cuts in (a reaction over an idle moment): it replaces the one playing, at once; what waits keeps waiting. */
export function interrupt<P extends string>(p: PlayerState<P>, anim: Anim<P>, now: number): PlayerState<P> {
  return { ...p, over: { anim, at: now } }
}

/** Ends what has played out by `now` and starts what waits, each at the end of the one before; the same object when nothing ended. */
export function advance<P extends string>(p: PlayerState<P>, now: number): PlayerState<P> {
  let s = p
  while (s.over && now >= s.over.at + animLength(s.over.anim)) {
    const end = s.over.at + animLength(s.over.anim)
    const [next, ...rest] = s.queue
    s = { ...s, over: next ? { anim: next, at: end } : null, queue: rest }
  }
  return s
}

/** The pose showing at `now`: the one-shot's while one plays, else the base loop's. */
export function poseAt<P extends string>(p: PlayerState<P>, now: number): P {
  if (p.over && now >= p.over.at) return stepAt(p.over.anim, now - p.over.at).pose
  return stepAt(p.base, now - p.baseAt).pose
}

/** Everything played out at once (the window shown again, a settle): the base alone, from `now`. */
export function settle<P extends string>(p: PlayerState<P>, now: number): PlayerState<P> {
  if (!p.over && !p.queue.length) return p
  return { base: p.base, baseAt: now, over: null, queue: [] }
}
```

```ts
// The idle schedule (0.11.0): when a character's next idle moment comes and which one, seeded by the game day, the
// character and how many moments it has had that day. Pure integer hashing: no Math.random, no clock.

/** The shortest wait between two idle moments, and how much longer it may be (ms). */
export const IDLE_MIN_MS = 6000
export const IDLE_SPAN_MS = 8000

/** Three integers hashed to one unsigned 32-bit integer (a salted mix, then a murmur3-style finaliser). */
export function mix(a: number, b: number, c: number): number {
  let h = 0x2545f491 ^ Math.imul(a | 0, 0x9e3779b1) ^ Math.imul(b | 0, 0x85ebca77) ^ Math.imul(c | 0, 0xc2b2ae3d)
  h = Math.imul(h ^ (h >>> 16), 0x7feb352d)
  h = Math.imul(h ^ (h >>> 15), 0x846ca68b)
  return (h ^ (h >>> 16)) >>> 0
}

/**
 * The n-th idle moment of the day for character `who` (0 the mangaka, 1-3 the assistants' seats): the wait before it,
 * IDLE_MIN_MS to IDLE_MIN_MS + IDLE_SPAN_MS - 1, and a pick the caller takes modulo its pool's length.
 */
export function nextIdle(day: number, who: number, n: number): { gapMs: number; pick: number } {
  const h = mix(day, who, n)
  return { gapMs: IDLE_MIN_MS + (h % IDLE_SPAN_MS), pick: h >>> 16 }
}
```

```ts
// The scene's reactions (0.11.0): which game event a store change brings that the cast reacts to. StudioScene
// subscribes and hands each one to the snapshot as a one-shot cue (`Snapshot.reaction`). Pure: reads two states,
// touches nothing. Good and bad are the corkboard's own tones (story.ts `rankToneOf`).
import { chapterDeadline } from '../../sim'
import type { GameState } from '../../sim'
import type { PopState } from './Pops'
import { rankToneOf } from './story'

export type ReactionKind = 'good-poll' | 'bad-poll' | 'level-up' | 'late' | 'missed'

/** A reaction for the snapshot: `id` grows with every cue, so each character plays it once. */
export interface ReactionCue { readonly id: number; readonly kind: ReactionKind }

/** Issues missed by every serial, running or archived (an archived serial keeps its count). */
export function missedIssuesOf(g: GameState): number {
  let n = g.serial ? g.serial.missedIssues : 0
  for (const s of g.serials) n += s.missedIssues
  return n
}

/** The newest poll a change brought: a one-shot result dismissed (so seen), or a new chapter result of the serial running before. */
function newPoll(a: GameState, b: GameState): { rank: number; entrants: number } | null {
  if (a.pendingResult && !b.pendingResult) return a.pendingResult
  const ser = a.serial
  if (!ser) return null
  const now = b.serial?.id === ser.id ? b.serial : b.serials.find((x) => x.id === ser.id)
  if (!now || now.results.length <= ser.results.length) return null
  return now.results[now.results.length - 1]
}

/**
 * The reaction one store change brings, or null: never across a save load, a new career (another world seed), a jump
 * of more than a day, or in a read-only tab (popsFor's guards). By priority: a missed issue, a bad poll, a good poll,
 * a level-up, the chapter turning late.
 */
export function reactionFor(prev: PopState, next: PopState): ReactionKind | null {
  const a = prev.game
  const b = next.game
  // most store changes (a modal, a toast, the speed) leave the game object as it was
  if (a === b) return null
  if (prev.readOnly || next.readOnly || prev.saveLoads !== next.saveLoads || !a || !b) return null
  if (a.worldSeed !== b.worldSeed) return null
  const gap = b.day - a.day
  if (gap !== 0 && gap !== 1) return null
  if (missedIssuesOf(b) > missedIssuesOf(a)) return 'missed'
  const poll = newPoll(a, b)
  const tone = poll ? rankToneOf(poll.rank, poll.entrants) : 'none'
  if (tone === 'low') return 'bad-poll'
  if (tone === 'top') return 'good-poll'
  if (b.player.level > a.player.level) return 'level-up'
  if (chapterDeadline(a)?.status !== 'late' && chapterDeadline(b)?.status === 'late') return 'late'
  return null
}
```

```ts
/**
 * What is wrong with the head and its face drawings, [] when nothing: the eye and mouth slots hold only face keys and
 * already hold the open eyes and the closed mouth (the bare head is neutral); the mark slot is clear; every eyes and
 * mouth drawing fills its slot (its size, face keys only, no '.'); every mark is the mark slot's size in mark keys.
 */
export function headRules(head: HeadPart, art: FaceArt): string[] {
  const out: string[] = []
  const window = (s: Slot) => head.map.slice(s.y, s.y + s.h).map((row) => row.slice(s.x, s.x + s.w))
  const { eyes, mouth, mark } = head.slots
  for (const [name, s] of [['eye', eyes], ['mouth', mouth]] as const) {
    window(s).forEach((row, r) => {
      for (let c = 0; c < row.length; c++) if (!FACE_KEYS.includes(row[c])) out.push(`${name} slot row ${s.y + r} col ${s.x + c} is ${row[c]}, not a face key`)
    })
  }
  window(mark).forEach((row, r) => {
    for (let c = 0; c < row.length; c++) if (row[c] !== '.') out.push(`mark slot row ${mark.y + r} col ${mark.x + c} is ${row[c]}, not clear`)
  })
  if (window(eyes).join('/') !== art.eyes.open.join('/')) out.push('the eye slot does not hold the open eyes')
  if (window(mouth).join('/') !== art.mouths.closed.join('/')) out.push('the mouth slot does not hold the closed mouth')
  const sized = (what: string, map: SpriteMap, s: Slot, keys: string) => {
    const w = map[0]?.length ?? 0
    if (map.length !== s.h || map.some((row) => row.length !== s.w)) out.push(`${what} is ${w}x${map.length}, not ${s.w}x${s.h}`)
    for (const ch of new Set(map.join(''))) if (!keys.includes(ch)) out.push(`${what} has key ${ch}`)
  }
  for (const [id, map] of Object.entries(art.eyes)) sized(`eyes ${id}`, map, eyes, FACE_KEYS)
  for (const [id, map] of Object.entries(art.mouths)) sized(`mouth ${id}`, map, mouth, FACE_KEYS)
  for (const [id, map] of Object.entries(art.marks)) sized(`mark ${id}`, map, mark, `.${MARK_KEYS}`)
  return out
}
```

```ts
// The rig's frames as the sprite cache draws them (0.11.0): a pose with the face its mood picks, assembled on first use
// and memoized, so render looks each one up by pose and mood and allocates nothing once it is built.
import { assemblePose } from './assemble'
import { FACE_ART, FACE_BY_MOOD } from './head'
import { PARTS } from './parts'
import { POSES } from './poses'
import type { PoseId } from './poses'
import type { CastFrame, Face, Mood, PartPose } from './types'

export const MOODS: readonly Mood[] = ['neutral', 'happy', 'tired', 'surprised', 'sleepy']
const MOOD_INDEX: Readonly<Record<Mood, number>> = { neutral: 0, happy: 1, tired: 2, surprised: 3, sleepy: 4 }

/** A part pose's face: the mood's, the pose's own slots over it, and no mark while the manuscript is ready (the '!' goes there). */
export function faceOf(pose: PartPose, mood: Mood, ready: boolean): Face {
  const face: Face = { ...FACE_BY_MOOD[mood], ...pose.face }
  return ready ? { ...face, mark: 'none' } : face
}

/** A frame's id: a key pose's own id; a part pose's with its face, `stand.idle~open.closed.none`. */
export function frameId(pose: PoseId, face: Face | null): string {
  return face ? `${pose}~${face.eyes}.${face.mouth}.${face.mark}` : pose
}

const memo = new Map<PoseId, (CastFrame | undefined)[]>()

/** The cast frame for a pose in a mood, ready or not: built on first use, the same object every time after. */
export function rigFrame(id: PoseId, mood: Mood, ready: boolean): CastFrame {
  let row = memo.get(id)
  if (!row) {
    row = []
    memo.set(id, row)
  }
  const pose = POSES[id].pose
  const k = pose.kind === 'key' ? 0 : MOOD_INDEX[mood] * 2 + (ready ? 1 : 0)
  let f = row[k]
  if (!f) {
    if (pose.kind === 'key') f = { id: frameId(id, null), map: pose.map }
    else {
      const face = faceOf(pose, mood, ready)
      f = { id: frameId(id, face), map: assemblePose(pose, PARTS, FACE_ART, face) }
    }
    row[k] = f
  }
  return f
}
```

```ts
// The cast driver (0.11.0): each loop step it turns every rig-drawn actor's state (actors.ts) into a pose on the actor
// (`Actor.pose`) through the animation player: the base loop for the state, the transition between two states, the
// idle moments the seeded schedule picks and the reactions the snapshot's cue brings. The tracks are mutated in
// place, like the actor table; the rules they follow are pure functions, tested alone. No clock: StudioScene passes
// its scene time.
import { LOW_STAMINA_THRESHOLD } from '../../sim'
import { RIG_LIVE } from './actors'
import type { Actor, ActorState, ActorTable } from './actors'
import { advance, enqueue, interrupt, poseAt, settle, startPlayer, switchBase } from './animation'
import type { PlayerState } from './animation'
import { nextIdle } from './idleSchedule'
import type { ReactionCue, ReactionKind } from './reactions'
import { ANIMS } from './rig/anims'
import type { AnimId } from './rig/anims'
import type { PoseId } from './rig/poses'

export type CastWho = 'mangaka' | 'assistant'

/** What picks a base loop besides the state: drawing with the clock running, the chapter late, the manuscript ready. */
export interface CastContext { working: boolean; late: boolean; ready: boolean }

/** The base loop for a character in a state, or null when it is not drawn (away, hidden, the cat's sitting). */
export function baseAnimFor(who: CastWho, state: ActorState, c: CastContext): AnimId | null {
  switch (state) {
    case 'seated': return c.working ? (c.late && who === 'mangaka' ? 'seat-draw-late' : 'seat-draw') : 'seat-idle'
    case 'standing': return who === 'mangaka' ? (c.ready ? 'stand-ready' : 'stand') : null
    case 'walking': return who === 'mangaka' ? 'walk' : null
    case 'sleeping': return who === 'mangaka' ? 'sleep' : 'doze'
    default: return null
  }
}

/** The transition between two states, or null: down into the chair and up out of it, into bed and out of it. */
export function transitionFor(from: ActorState, to: ActorState): AnimId | null {
  if (from === 'walking' && to === 'seated') return 'sit-down'
  if (from === 'seated' && to === 'walking') return 'stand-up'
  if (from === 'walking' && to === 'sleeping') return 'lie-down'
  if (from === 'sleeping' && to === 'walking') return 'get-up'
  return null
}

const STAND_POOL: readonly AnimId[] = ['stand-look', 'stand-sip']
const SEAT_POOL: readonly AnimId[] = ['seat-stretch', 'seat-sip', 'seat-flip']
const TIRED_POOL: readonly AnimId[] = ['seat-rub', 'seat-sip']
const NO_POOL: readonly AnimId[] = []

/** The idle moments a base loop picks from: standing, a look around or a sip; at the desk, a stretch, a sip or a page flipped, or, tired, rubbing the eyes. */
export function idlePool(base: AnimId, tired: boolean): readonly AnimId[] {
  if (base === 'stand') return STAND_POOL
  if (base === 'seat-draw' || base === 'seat-idle') return tired ? TIRED_POOL : SEAT_POOL
  return NO_POOL
}

/** Every idle moment (the pools' animations): a reaction cuts one short; it waits behind a transition or another reaction. */
export const IDLE_MOMENTS: ReadonlySet<string> = new Set<AnimId>([...STAND_POOL, ...SEAT_POOL, ...TIRED_POOL])

/** The animation a reaction plays for a character in a state, or null (the plan's decision 7). */
export function reactionAnim(kind: ReactionKind, who: CastWho, state: ActorState): AnimId | null {
  const seated = state === 'seated'
  const standing = state === 'standing'
  if (who === 'assistant') return kind === 'good-poll' && seated ? 'seat-cheer' : null
  switch (kind) {
    case 'good-poll':
    case 'level-up': return standing ? 'stand-cheer' : seated ? 'seat-cheer' : null
    case 'bad-poll': return standing ? 'stand-slump' : seated ? 'seat-slump' : null
    case 'late': return seated ? 'seat-panic' : null
    case 'missed': return seated ? 'seat-tear' : null
  }
}

/** The rig head's top row in a state's frame: 6 seated, 8 in the bed frame, else 0 (bubbles and pops sit above it). */
export function headTop(state: ActorState): number {
  return state === 'seated' ? 6 : state === 'sleeping' ? 8 : 0
}

/** The rig head's centre column in a state's frame: 16 (cols 4-27), 35 in the bed frame (cols 23-46), drawn at the bed. */
export function headCentre(state: ActorState): number {
  return state === 'sleeping' ? 35 : 16
}

/** One character's animation: its player, the state it last saw, the last cue it took, its idle schedule. */
export interface Track {
  player: PlayerState<PoseId> | null
  state: ActorState
  cue: number
  /** The game day the count is for, and the moments counted that day (the schedule's seed). */
  idleDay: number
  idleN: number
  /** When the next idle moment fires, on the scene clock; -1 until it is scheduled. A new day never moves it. */
  nextIdleAt: number
}

export interface CastPlay { readonly mangaka: Track; readonly assistants: readonly [Track, Track, Track] }

function newTrack(): Track {
  return { player: null, state: 'hidden', cue: 0, idleDay: -1, idleN: 0, nextIdleAt: -1 }
}

export function newCastPlay(): CastPlay {
  return { mangaka: newTrack(), assistants: [newTrack(), newTrack(), newTrack()] }
}

/** What the driver reads each step besides the actor table (StudioScene fills one object in place). */
export interface CastInput {
  day: number
  working: boolean
  late: boolean
  ready: boolean
  stamina: number
  assistants: readonly { stamina: number }[]
  reaction: ReactionCue | null
  /** A dev build's `?rig=<animation>`: played over every base of its posture whenever nothing else plays. */
  preview: AnimId | null
}

/** How far apart the assistants' loops start, so their pens never move in step. */
export const PHASE_MS = 150

const ctx: CastContext = { working: false, late: false, ready: false }

function baseOf(who: CastWho, a: Actor, input: CastInput): AnimId | null {
  ctx.working = input.working
  ctx.late = input.late
  ctx.ready = input.ready
  return baseAnimFor(who, a.state, ctx)
}

function stepTrack(t: Track, a: Actor, who: CastWho, seat: number, live: boolean, input: CastInput, now: number, reduced: boolean): void {
  if (!live) {
    a.pose = ''
    return
  }
  const id = baseOf(who, a, input)
  if (id === null) {
    t.player = null
    t.state = a.state
    a.pose = ''
    return
  }
  const base = ANIMS[id]
  if (!t.player) {
    t.player = startPlayer(base, now - seat * PHASE_MS)
    t.cue = input.reaction?.id ?? t.cue
  } else if (base !== t.player.base) {
    // a new base keeps the seat's phase (so the assistants' pens never fall into step); a transition starts now
    const tr = reduced ? null : transitionFor(t.state, a.state)
    t.player = switchBase(t.player, base, tr === null ? null : ANIMS[tr], tr === null ? now - seat * PHASE_MS : now)
  }
  t.state = a.state
  // end what played out since the last step first, so nothing below sees a finished one-shot as still playing
  t.player = advance(t.player, now)
  if (input.reaction && input.reaction.id !== t.cue) {
    t.cue = input.reaction.id
    const r = reduced ? null : reactionAnim(input.reaction.kind, who, a.state)
    if (r !== null) {
      // a reaction cuts an idle moment short; it waits behind a transition or another reaction
      const playing = t.player.over?.anim.id
      t.player = playing !== undefined && IDLE_MOMENTS.has(playing) ? interrupt(t.player, ANIMS[r], now) : enqueue(t.player, ANIMS[r], now)
    }
  }
  if (reduced) {
    a.pose = base.steps[0].pose
    return
  }
  // a new game day restarts the count (the schedule's seed), never the wait: days turn every 220-1000 ms while the
  // clock runs, and the shortest wait is 6 s
  if (t.idleDay !== input.day) {
    t.idleDay = input.day
    t.idleN = 0
  }
  if (t.nextIdleAt < 0) t.nextIdleAt = now + nextIdle(input.day, seat, t.idleN).gapMs
  if (now >= t.nextIdleAt) {
    const stamina = who === 'mangaka' ? input.stamina : input.assistants[seat - 1]?.stamina ?? 100
    const pool = idlePool(id, stamina < LOW_STAMINA_THRESHOLD)
    if (pool.length && !t.player.over) t.player = enqueue(t.player, ANIMS[pool[nextIdle(input.day, seat, t.idleN).pick % pool.length]], now)
    t.idleN++
    t.nextIdleAt = now + nextIdle(input.day, seat, t.idleN).gapMs
  }
  const preview = input.preview === null ? null : ANIMS[input.preview]
  if (preview && preview !== base && preview.posture === base.posture && !t.player.over) t.player = enqueue(t.player, preview, now)
  a.pose = poseAt(t.player, now)
}

/** One loop step for the whole cast at scene time `now`; `reduced` (reduced motion) holds every base's first step. */
export function stepCast(play: CastPlay, table: ActorTable, input: CastInput, now: number, reduced: boolean): void {
  stepTrack(play.mangaka, table.mangaka, 'mangaka', 0, RIG_LIVE.mangaka, input, now, reduced)
  for (let i = 0; i < 3; i++) stepTrack(play.assistants[i], table.assistants[i], 'assistant', i + 1, RIG_LIVE.assistants, input, now, reduced)
}

function settleTrack(t: Track, a: Actor, who: CastWho, seat: number, live: boolean, input: CastInput, now: number): void {
  if (!live) {
    a.pose = ''
    return
  }
  const id = baseOf(who, a, input)
  t.player = id === null ? null : startPlayer(ANIMS[id], now - seat * PHASE_MS)
  t.state = a.state
  t.cue = input.reaction?.id ?? t.cue
  t.idleDay = -1
  t.nextIdleAt = -1
  a.pose = id === null ? '' : ANIMS[id].steps[0].pose
}

/** Everyone on their base loop from now (a mount, a loaded save): no transition, the waiting cue taken as seen. */
export function settleCast(play: CastPlay, table: ActorTable, input: CastInput, now: number): void {
  settleTrack(play.mangaka, table.mangaka, 'mangaka', 0, RIG_LIVE.mangaka, input, now)
  for (let i = 0; i < 3; i++) settleTrack(play.assistants[i], table.assistants[i], 'assistant', i + 1, RIG_LIVE.assistants, input, now)
}

/** The window shown again: every transition and moment jumps to its end, each base going on from now. */
export function resumeCast(play: CastPlay, now: number): void {
  for (const t of [play.mangaka, ...play.assistants]) if (t.player) t.player = settle(t.player, now)
}
```
